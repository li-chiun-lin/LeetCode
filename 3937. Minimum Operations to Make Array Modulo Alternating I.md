# Medium

You are given an integer array $nums$ and an integer $k$.

In one operation, you can increase or decrease any element of $nums$ by 1.

An array is called modulo alternating if there exist two distinct integers $x$ and $y$ ($0 \leq x, y < k$) such that:

- For every even index $i$, $nums[i] \mod k == x$
- For every odd index $i$, $nums[i] \mod k == y$

Return the minimum number of operations required to make $nums$ modulo alternating.

```cpp
class Solution {
    tuple<int, int, int, int> count(vector<int>& nums, int i, int k)
    {
        int n = size(nums);

        vector<int> c(k);
        int sum = 0;
        int m = 0;

        while (i < n)
        {
            int d = nums[i];
            ++ c[d];
            sum += d;
            i += 2;
            ++ m;
        }

        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> pq;

        for (int j = 0; j < k; ++j)
        {
            int sum = 0;
            
            for (int a = 0; a < k; ++a)
            {
                int d = abs(j - a);
                sum += min(d, k - d) * c[a];
            }

            pq.push({sum, j});
        }

        auto [s1, v1] = pq.top();
        pq.pop();
        auto [s2, v2] = pq.top();

        return {s1, v1, s2, v2};
    }
    
public:
    int minOperations(vector<int>& nums, int k) {
        if (size(nums) == 1)
            return 0;

        for (auto& x : nums)
            x %= k;
        
        auto [a1, c1, a2, c2] = count(nums, 0, k);
        auto [a3, c3, a4, c4] = count(nums, 1, k);

        if (c1 == c3)
            return min(a1 + a4, a2 + a3);
        
        return a1 + a3;
    }
};
```
