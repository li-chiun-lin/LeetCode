# Hard

You are given a rooted tree with $n$ nodes labeled from $0$ to $n - 1$, represented by an integer array $parent$ of length $n$, where:

- $parent[0] = -1$ (node 0 is the root).
- For each $1 \leq i < n$, $parent[i]$ is the parent of node i ($0 \leq parent[i] < i$).

You are also given an integer array $nums$ of length $n$, where $nums[i]$ is the value of node $i$, and an integer $k$.

A non-empty subset of nodes is called valid if:

- The sum of the values of the selected nodes is divisible by $k$.
- No two selected nodes are adjacent in the tree (no node and its direct parent are both included in the subset).

Return the number of valid subsets modulo $10^9 + 7$.

```cpp
class Solution {
    static constexpr int MOD = 1e9 + 7;

    vector<vector<long long>> dfs(vector<int>& nums, int k, int u, vector<vector<int>>& g)
    {
        vector<vector<long long>> dp(2, vector<long long>(k));

        dp[0][0] = 1;
        dp[1][nums[u] % k] = 1;

        for (int v : g[u])
        {
            auto child = dfs(nums, k, v, g);

            vector<vector<long long>> ndp(2, vector<long long>(k));

            // u not selected
            for (int r1 = 0; r1 < k; ++r1)
            {
                if (dp[0][r1] == 0)
                {
                    continue;
                }

                for (int r2 = 0; r2 < k; ++r2)
                {
                    long long ways =
                        (child[0][r2] + child[1][r2]) % MOD;

                    int nr = (r1 + r2) % k;

                    ndp[0][nr] =
                        (ndp[0][nr] +
                         dp[0][r1] * ways) % MOD;
                }
            }

            // u selected
            for (int r1 = 0; r1 < k; ++r1)
            {
                if (dp[1][r1] == 0)
                {
                    continue;
                }

                for (int r2 = 0; r2 < k; ++r2)
                {
                    if (child[0][r2] == 0)
                    {
                        continue;
                    }

                    int nr = (r1 + r2) % k;

                    ndp[1][nr] =
                        (ndp[1][nr] +
                         dp[1][r1] * child[0][r2]) % MOD;
                }
            }

            dp = move(ndp);
        }

        return dp;
    }

public:
    int countValidSubsets(vector<int>& parent, vector<int>& nums, int k) {
        int n = parent.size();
        vector<vector<int>> g(n);

        for (int i = 1; i < n; ++i)
        {
            g[parent[i]].push_back(i);
        }

        auto dp = dfs(nums, k, 0, g);

        long long ans =
            (dp[0][0] + dp[1][0] - 1 + MOD) % MOD;

        return ans;
    }
};
```
