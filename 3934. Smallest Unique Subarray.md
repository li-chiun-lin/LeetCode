# Hard

You are given an integer array $nums$.

Find the minimum length of a subarray that is not identical to any other subarray in $nums$.

Return an integer denoting the minimum possible length of such a subarray.

```cpp
class Solution {
    // Large prime numbers for Double Rolling Hash to avoid collisions
    const long long MOD1 = 1e9 + 7;
    const long long MOD2 = 1e9 + 9;
    const long long BASE1 = 313;
    const long long BASE2 = 317;

    bool hasUniqueSubarray(const vector<int>& nums, int L) {
        int n = nums.size();
        
        // Maps a pair of hashes to its frequency count
        // Using a custom hash or a nested structure for pairs
        auto pair_hash = [](const pair<long long, long long>& p) {
            return (p.first << 32) ^ p.second;
        };
        unordered_map<pair<long long, long long>, int, decltype(pair_hash)> hash_counts(0, pair_hash);

        long long h1 = 0, h2 = 0;
        long long p1 = 1, p2 = 1;

        // Compute the multiplier for the leading element
        for (int i = 0; i < L - 1; ++i) {
            p1 = (p1 * BASE1) % MOD1;
            p2 = (p2 * BASE2) % MOD2;
        }

        // Compute hash for the first window
        for (int i = 0; i < L; ++i) {
            h1 = (h1 * BASE1 + nums[i]) % MOD1;
            h2 = (h2 * BASE2 + nums[i]) % MOD2;
        }
        hash_counts[{h1, h2}]++;

        // Slide the window across the rest of the array
        for (int i = L; i < n; ++i) {
            // Remove the out-going element
            h1 = (h1 - nums[i - L] * p1) % MOD1;
            if (h1 < 0) h1 += MOD1;
            
            h2 = (h2 - nums[i - L] * p2) % MOD2;
            if (h2 < 0) h2 += MOD2;

            // Add the incoming element
            h1 = (h1 * BASE1 + nums[i]) % MOD1;
            h2 = (h2 * BASE2 + nums[i]) % MOD2;

            hash_counts[{h1, h2}]++;
        }

        // Check if any subarray hash appeared exactly once
        for (auto& [hash_pair, count] : hash_counts) {
            if (count == 1) return true;
        }

        return false;
    }

public:
    int smallestUniqueSubarray(vector<int>& nums) {
        int n = nums.size();
        int low = 1, high = n;
        int ans = n;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (hasUniqueSubarray(nums, mid)) {
                ans = mid;       // This length works, try to find a smaller one
                high = mid - 1;
            } else {
                low = mid + 1;   // Need a larger length to find uniqueness
            }
        }

        return ans;
    }
};
```
