# Medium

You are given an $n \times m$ integer matrix $matrix$ containing non-negative integers.

A non-zero cell $(row, col)$ checks the cells near it as follows:

- Let $x = matrix[row][col]$.
- Consider every cell within $x$ rows and $x$ columns of $(row, col)$.
- Ignore cells that are outside the matrix.
- Ignore the cells where both the row distance and column distance are exactly $x$.

The cell $(row, col)$ is a local maximum if it is non-zero and no considered cell has a value greater than $x$.

Return an integer denoting the number of local maximums in $matrix$.

```cpp
class Solution {
public:
    int countLocalMaximums(vector<vector<int>>& matrix) {
        int n = matrix.size();
        int m = matrix[0].size();
        
        int logN = log2(n) + 1;
        int logM = log2(m) + 1;
        
        // st[i][j][r][c] stores the max in a block of size (2^i) x (2^j) starting at (r, c)
        vector<vector<vector<vector<int>>>> st(logN, vector<vector<vector<int>>>(logM, 
            vector<vector<int>>(n, vector<int>(m, 0))));
            
        // Base case: blocks of size 1x1
        for (int r = 0; r < n; ++r) {
            for (int c = 0; c < m; ++c) {
                st[0][0][r][c] = matrix[r][c];
            }
        }
        
        // Build the 2D Sparse Table
        for (int i = 0; i < logN; ++i) {
            for (int j = 0; j < logM; ++j) {
                if (i == 0 && j == 0) continue;
                for (int r = 0; r + (1 << i) <= n; ++r) {
                    for (int c = 0; c + (1 << j) <= m; ++c) {
                        if (i > 0) {
                            st[i][j][r][c] = max(st[i-1][j][r][c], st[i-1][j][r + (1 << (i-1))][c]);
                        } else {
                            st[i][j][r][c] = max(st[i][j-1][r][c], st[i][j-1][r][c + (1 << (j-1))]);
                        }
                    }
                }
            }
        }
        
        // Lambda helper to query max in rectangle [r1, r2] x [c1, c2]
        auto queryMax = [&](int r1, int c1, int r2, int c2) {
            if (r1 > r2 || c1 > c2) return 0;
            int i = log2(r2 - r1 + 1);
            int j = log2(c2 - c1 + 1);
            
            return max({
                st[i][j][r1][c1],
                st[i][j][r2 - (1 << i) + 1][c1],
                st[i][j][r1][c2 - (1 << j) + 1],
                st[i][j][r2 - (1 << i) + 1][c2 - (1 << j) + 1]
            });
        };

        int localMaxCount = 0;

        for (int r = 0; r < n; ++r) {
            for (int c = 0; c < m; ++c) {
                int x = matrix[r][c];
                if (x == 0) continue;

                // Bounding box boundaries
                int r1 = max(0, r - x), r2 = min(n - 1, r + x);
                int c1 = max(0, c - x), c2 = min(m - 1, c + x);

                int maxInRegion = 0;

                // Instead of querying the whole box and excluding the corners,
                // we can break the valid region into 3 horizontal strips:
                // 1. Top row (r - x): Excludes the corners (c - x) and (c + x)
                // 2. Middle rows (r - x + 1 to r + x - 1): Full width from c1 to c2
                // 3. Bottom row (r + x): Excludes the corners (c - x) and (c + x)

                // 1. Top row (if within bounds)
                if (r - x >= 0) {
                    int top_c1 = max(0, c - x + 1);
                    int top_c2 = min(m - 1, c + x - 1);
                    maxInRegion = max(maxInRegion, queryMax(r - x, top_c1, r - x, top_c2));
                }

                // 2. Bottom row (if within bounds)
                if (r + x < n) {
                    int bot_c1 = max(0, c - x + 1);
                    int bot_c2 = min(m - 1, c + x - 1);
                    maxInRegion = max(maxInRegion, queryMax(r + x, bot_c1, r + x, bot_c2));
                }

                // 3. Middle body
                int mid_r1 = max(0, r - x + 1);
                int mid_r2 = min(n - 1, r + x - 1);
                maxInRegion = max(maxInRegion, queryMax(mid_r1, c1, mid_r2, c2));

                // If no considered cell is strictly greater than x, it's a local maximum
                if (maxInRegion <= x) {
                    localMaxCount++;
                }
            }
        }

        return localMaxCount;
    }
};
```
