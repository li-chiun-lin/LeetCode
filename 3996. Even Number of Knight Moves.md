# Easy

You are given two integer arrays $start$ and $target$, where each array is of the form $[x, y]$ representing a cell on a standard $8 \times 8$ chessboard.

Return $true$ if a knight can move from $start$ to $target$ in an even number of moves. Otherwise, return $false$.

```cpp
class Solution {
public:
    bool canReach(vector<int>& start, vector<int>& target) {
        vector<vector<bool>> visited(8, vector<bool>(8, false));
        int move = 0;
        vector<int> d = {1, 2, -1, 2, 1, -2, -1, -2, 1};
        queue<pair<int, int>> que;
        que.push({start[0], start[1]});
        visited[start[0]][start[1]] = true;

        while (int n = size(que))
        {
            while (n --)
            {
                auto [x, y] = que.front();
                que.pop();

                if (x == target[0] && y == target[1] && move % 2 == 0)
                    return true;

                for (int k = 1; k < 8; ++k)
                {
                    int xx = x + d[k];
                    int yy = y + d[k + 1];

                    if (0 <= xx && xx < 8 && 0 <= yy && yy < 8 && ! visited[xx][yy])
                    {
                        visited[xx][yy] = true;
                        que.push({xx, yy});
                    }
                }
            }

            ++ move;
        }

        return false;
    }
};
```
